C++17 Algorithm Explorer

A menu-driven console program, written in standard C++17, that shows the algorithms it runs step by step. It runs on your machine, not in the browser.

  • Binary search trace
  • 5 sorting algorithms compared
  • BFS & DFS traversal
  • Dijkstra shortest paths
  • LCS dynamic programming
  • Sieve of Eratosthenes

// portfolio.cpp: C++17 Algorithm Explorer
#include <vector>
#include <queue>

namespace explorer {

// Dijkstra with a min-heap: O((V + E) log V)
std::vector<long long> dijkstra(const Graph& g, int src) {
    std::vector<long long> dist(g.size(), INF);
    std::priority_queue<Item, std::vector<Item>,
                        std::greater<Item>> pq;
    dist[src] = 0;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;
        for (const auto& [v, w] : g[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}

} // namespace explorer
